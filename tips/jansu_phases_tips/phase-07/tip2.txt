## 4. Add full Produce size validation

`ProduceService::validate_batch` already validates magic, timestamps, record count, last offset delta, offset deltas, CRC, and canonical batch length.  It does not yet enforce Kafka-style message size limits.

### Add limits

```rust
// jansu-storage/src/service/produce.rs

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
struct ProduceLimits {
    max_record_bytes: usize,
    max_batch_bytes: usize,
}

impl Default for ProduceLimits {
    fn default() -> Self {
        Self {
            // Kafka broker default is around 1 MiB; keep this configurable later.
            max_record_bytes: 1_048_576,
            max_batch_bytes: 1_048_576,
        }
    }
}
```

### Extend validation

```rust
use jansu_sans_io::ByteSize;

fn validate_batch(batch: &deflated::Batch, limits: ProduceLimits) -> Result<()> {
    let _compression =
        BatchAttribute::try_from(batch.attributes).map_err(|error| match error {
            jansu_sans_io::Error::UnknownCompressionType(_) => {
                Error::Api(ErrorCode::UnsupportedCompressionType)
            }
            otherwise => Error::from(otherwise),
        })?;

    if batch.magic != Self::MAGIC_V2 {
        return Err(Error::Api(ErrorCode::InvalidRecord));
    }

    let batch_size = batch.size_in_bytes()?;
    if batch_size > limits.max_batch_bytes {
        return Err(Error::Api(ErrorCode::MessageTooLarge));
    }

    if batch.base_timestamp < 0
        || batch.max_timestamp < 0
        || batch.max_timestamp < batch.base_timestamp
    {
        return Err(Error::Api(ErrorCode::InvalidTimestamp));
    }

    let records: Vec<Record> = batch.try_into().map_err(|error| {
        debug!(?error);
        Error::Api(ErrorCode::CorruptMessage)
    })?;

    if records.is_empty() {
        return Err(Error::Api(ErrorCode::InvalidRecord));
    }

    for record in &records {
        if record.size_in_bytes()? > limits.max_record_bytes {
            return Err(Error::Api(ErrorCode::MessageTooLarge));
        }
    }

    if records
        .iter()
        .enumerate()
        .any(|(index, record)| record.offset_delta != index as i32)
    {
        return Err(Error::Api(ErrorCode::InvalidRecord));
    }

    let expected_record_count =
        u32::try_from(records.len()).map_err(|_| Error::Api(ErrorCode::InvalidRecord))?;
    if batch.record_count != expected_record_count {
        return Err(Error::Api(ErrorCode::InvalidRecord));
    }

    let expected_last_offset_delta =
        i32::try_from(records.len() - 1).map_err(|_| Error::Api(ErrorCode::InvalidRecord))?;
    if batch.last_offset_delta != expected_last_offset_delta {
        return Err(Error::Api(ErrorCode::InvalidRecord));
    }

    let inflated = inflated::Batch {
        base_offset: batch.base_offset,
        batch_length: batch.batch_length,
        partition_leader_epoch: batch.partition_leader_epoch,
        magic: batch.magic,
        crc: batch.crc,
        attributes: batch.attributes,
        last_offset_delta: batch.last_offset_delta,
        base_timestamp: batch.base_timestamp,
        max_timestamp: batch.max_timestamp,
        producer_id: batch.producer_id,
        producer_epoch: batch.producer_epoch,
        base_sequence: batch.base_sequence,
        records,
    };

    let canonical = deflated::Batch::try_from(inflated).map_err(|error| {
        debug!(?error);
        Error::Api(ErrorCode::CorruptMessage)
    })?;

    if canonical.batch_length != batch.batch_length || canonical.crc != batch.crc {
        return Err(Error::Api(ErrorCode::CorruptMessage));
    }

    Ok(())
}
```

## 5. Add Snappy support

The Phase 07 spec requires gzip, snappy, lz4, and zstd. The current deflated record code already has gzip, lz4, and zstd encode paths; `snap` is already a dependency of `jansu-sans-io`, but the encode path currently treats unsupported compression as unexpected.  

### Add Snappy encode

```rust
// jansu-sans-io/src/record/deflated.rs

fn into_record_data(records: &[Record], compression: Compression) -> Result<Bytes> {
    let sizing = records
        .iter()
        .map(|record| record.size_in_bytes())
        .collect::<Result<Vec<_>>>()?
        .iter()
        .sum::<usize>();

    debug!(sizing);

    match compression {
        Compression::None => records.encode(),

        Compression::Gzip => {
            let uncompressed = records.encode()?;
            let mut gz = GzEncoder::new(
                BytesMut::with_capacity(uncompressed.len()).writer(),
                flate2::Compression::default(),
            );
            gz.write_all(&uncompressed)?;
            gz.finish()
                .map(|w| w.into_inner())
                .map(Bytes::from)
                .map_err(Into::into)
        }

        Compression::Snappy => {
            let uncompressed = records.encode()?;
            let mut encoder =
                snap::write::FrameEncoder::new(BytesMut::with_capacity(uncompressed.len()).writer());
            encoder.write_all(&uncompressed)?;
            encoder
                .into_inner()
                .map(|w| Bytes::from(w.into_inner()))
                .map_err(Into::into)
        }

        Compression::Lz4 => {
            let uncompressed = records.encode()?;
            let mut lz4 = lz4::EncoderBuilder::new()
                .build(BytesMut::with_capacity(uncompressed.len()).writer())?;
            lz4.write_all(&uncompressed[..])?;
            let (w, _) = lz4.finish();
            Ok(Bytes::from(w.into_inner()))
        }

        Compression::Zstd => {
            let uncompressed = records.encode()?;
            let mut zstd = zstd::stream::write::Encoder::new(
                BytesMut::with_capacity(uncompressed.len()).writer(),
                0,
            )?;
            zstd.write_all(&uncompressed[..])?;
            zstd.finish()
                .map(|w| w.into_inner())
                .map(Bytes::from)
                .map_err(Into::into)
        }
    }
}
```

## 6. Make multi-batch Produce atomic per partition

Current `ProduceService::partition` validates all batches first, then appends them one by one. That prevents validation failures from partially appending, but it does not prevent a storage failure on the second or later batch from leaving earlier batches appended while the partition response reports failure. 

That is the biggest remaining correctness gap.

### Required storage trait addition

Add a default `produce_partition` method to `Storage`:

```rust
// jansu-storage/src/lib.rs

#[async_trait]
pub trait Storage: Debug + Send + Sync + 'static {
    async fn produce(
        &self,
        transaction_id: Option<&str>,
        topition: &Topition,
        batch: deflated::Batch,
    ) -> Result<i64>;

    async fn produce_partition(
        &self,
        transaction_id: Option<&str>,
        topition: &Topition,
        batches: Vec<deflated::Batch>,
    ) -> Result<i64> {
        if batches.is_empty() {
            return Err(Error::Api(ErrorCode::InvalidRecord));
        }

        if batches.len() == 1 {
            return self.produce(transaction_id, topition, batches.into_iter().next().unwrap()).await;
        }

        let combined = crate::batch::combine_for_partition_append(batches)?;
        self.produce(transaction_id, topition, combined).await
    }
}
```

## 7. Fix batcher offset correctness

`ProduceRequestBatcher` currently combines queued requests and returns the same base offset to every ticket owner. That is wrong if two client requests are coalesced: the second client must receive the base offset of its first record, not the combined batch’s base offset. 

### Required changes

Change the batching key to include producer epoch:

```rust
#[derive(Clone, Debug, Default, Eq, Hash, Ord, PartialEq, PartialOrd)]
struct TopitionProducer {
    topition: Topition,
    producer_id: i64,
    producer_epoch: i16,
}
```

Change `BatchRequest`:

```rust
#[derive(Clone, Debug, Default, Eq, Hash, Ord, PartialEq, PartialOrd)]
struct BatchRequest {
    id: Uuid,
    first_record_delta: i64,
    record_count: i64,
    batch: deflated::Batch,
}
```

Compute offsets when sending:

```rust
let mut next_delta = 0_i64;
let queued = queued
    .into_iter()
    .map(|mut request| {
        request.first_record_delta = next_delta;
        request.record_count = i64::from(request.batch.last_offset_delta) + 1;
        next_delta += request.record_count;
        request
    })
    .collect::<Vec<_>>();

let owners = queued
    .iter()
    .map(|request| (request.id, request.first_record_delta))
    .collect::<Vec<_>>();

let combined = combine(queued.into_iter().map(|request| request.batch).collect())?;

if let Some(combined) = combined {
    let combined_base_offset = self
        .storage
        .produce(transaction_id, topition, combined)
        .await?;

    self.responses.lock().map(|mut responses| {
        for (owner, delta) in owners {
            let owner_base_offset = combined_base_offset + delta;
            if let Some(BatchResponse::Waker(waker)) =
                responses.insert(owner, BatchResponse::Response(owner_base_offset))
            {
                waker.wake();
            }
        }
    })?;
}
```
